{{- $image := resources.Get .path -}} {{- if not $image -}} {{- errorf "Missing site photo: %s" .path -}} {{- else -}} {{- $maxWidth := int (math.Min $image.Width 1440) -}} {{- $widths := slice -}} {{- range slice 320 640 960 -}} {{- if lt . $maxWidth -}}{{- $widths = $widths | append . -}}{{- end -}} {{- end -}} {{- $widths = $widths | append $maxWidth -}} {{- $srcset := slice -}} {{- range $widths -}} {{- $resized := $image.Resize (printf "%dx webp q80" .) -}} {{- $srcset = $srcset | append (printf "%s %dw" $resized.RelPermalink $resized.Width) -}} {{- end -}} {{- $fallback := $image.Resize (printf "%dx webp q80" (int (math.Min 960 $maxWidth))) -}} {{- end -}}
